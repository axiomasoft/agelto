// --- Source: AiDocumentType.php ---
/**
 * Тип документа, определённый LLM-скринингом (шаг screening, docs/specs/v2/03-document-pipeline.md §3.4).
 * - Estimate: смета (ЛСР, КС-2, ведомость ресурсов, дефектная ведомость).
 * - TechSpec: техническое задание.
 * - ContractDraft: проект контракта/договора.
 * - Protocol: протокол (рассмотрения заявок, подведения итогов и т. п.).
 * - Unknown: тип не распознан.
 */
enum AiDocumentType: string
{
    case Estimate = 'estimate';
    case TechSpec = 'tech_spec';
    case ContractDraft = 'contract_draft';
    case Protocol = 'protocol';
    case Unknown = 'unknown';

    /**
     * Единственный носитель списка значений для check-constraint'ов оси типа документа
     * (§1.3 правило 1, root/document-type-axis.md) — миграции складывают IN(...) отсюда,
     * не восемью копиями литералов.
     *
     * @return list<string>
     */
    public static function values(): array
    {
        return array_map(static fn (self $case): string => $case->value, self::cases());
    }
}

// --- Source: ScreeningResult.php ---
/**
 * Ответ LLM на шаге screening (первичный анализ, docs/specs/v2/03-document-pipeline.md §3.4;
 * docs/specs/v2/08-estimate-extraction.md §3.5). Модель получает только первые N страниц
 * документа (D6) — это дешёвая классификация, не извлечение данных.
 * - document_type: тип документа — estimate/tech_spec/contract_draft/protocol/unknown.
 * - confidence: уверенность модели в классификации, от 0 (совсем не уверена) до 1 (полностью уверена).
 * - description: краткое резюме файла — что это, за какой период, какой объём.
 * - tags: метки документа (Город, ОС, Шифр, Вид работ, ФИО) — как в исходном v1-классификаторе.
 * - page_count_estimate: оценочное число страниц документа, если видно из среза.
 * - ignore_reason: если документ точно не подходит для дальнейшей обработки — короткая причина
 *   на русском (иначе null; нечитаемое или неясное — null, не придумывать причину).
 */
final class ScreeningResult extends Data
{
    /**
     * @param  string[]  $tags
     */
    public function __construct(
        public AiDocumentType $document_type,
        public float $confidence,
        public ?string $description = null,
        public array $tags = [],
        public ?int $page_count_estimate = null,
        public ?string $ignore_reason = null,
    ) {}
}
